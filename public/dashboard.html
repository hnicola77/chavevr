/************************************************************ * DASHBOARD – JAVASCRIPT PRINCIPAL * Adaptado para Backend de Tabela Única (Strings) ************************************************************/ let todasUnidades = []; let unidadesFiltradas = []; // Dados fixos (Hardcoded) para preencher os selects. const EMPREENDIMENTOS_FIXOS = [ "Residencial Horizonte", "New Jersey", "Honolulu", "Plaza de Espanha", "Plaza Valencia", "Boulevard Fecile", "Federico Fellini" ]; const BLOCOS_FIXOS = [ "Torre A", "Torre B", "Bloco 01", "Bloco 02", "Bloco 03", ]; /************************************************************ * LOAD INICIAL ************************************************************/ document.addEventListener("DOMContentLoaded", async () => { // 1. Garante que os selects de Empreendimento usem os valores definidos preencherSelectsIniciais(); // 2. Carrega todos os dados do servidor await carregarDados(); // 3. Aplica o filtro inicial (mostra tudo) aplicarFiltro(); // 4. Adiciona listener de evento para o filtro de Empreendimento document.getElementById("fEmpreendimento").addEventListener("change", preencherFiltroBloco); }); /************************************************************ * PREENCHER FILTROS ************************************************************/ function preencherSelectsIniciais() { // Preenche o fEmpreendimento com os dados fixos const selectEmp = document.getElementById("fEmpreendimento"); selectEmp.innerHTML = ''; EMPREENDIMENTOS_FIXOS.forEach(e => { selectEmp.innerHTML += ``; }); // Chama a função para preencher o Bloco, que inicialmente mostrará todos os blocos fixos preencherFiltroBloco(); } function preencherFiltroBloco() { const selectBloco = document.getElementById("fBloco"); // Reinicia o select de Bloco selectBloco.innerHTML = ''; // Preenche o filtro de bloco com base nos dados fixos BLOCOS_FIXOS.forEach(b => { selectBloco.innerHTML += ``; }); } /************************************************************ * CARREGAR DADOS DO SERVIDOR ************************************************************/ async function carregarDados() { try { const res = await fetch("/unidades"); todasUnidades = await res.json(); } catch (error) { console.error("Erro ao carregar dados para o Dashboard:", error); } } /************************************************************ * LÓGICA DE FILTRO E ATUALIZAÇÃO ************************************************************/ function aplicarFiltro() { const fEmp = document.getElementById("fEmpreendimento").value; const fBloco = document.getElementById("fBloco").value; const fSit = document.getElementById("fSituacao").value; unidadesFiltradas = todasUnidades.filter(u => { // Filtro por Empreendimento (string) const matchEmp = !fEmp || u.empreendimento === fEmp; // Filtro por Bloco (string) const matchBloco = !fBloco || u.bloco === fBloco; // Filtro por Situação (string) const matchSit = !fSit || u.situacao === fSit; return matchEmp && matchBloco && matchSit; }); atualizarDashboard(); } /************************************************************ * ATUALIZAR CARDS E TABELA ************************************************************/ function atualizarDashboard() { // --- 1. CALCULAR TOTAIS --- const total = unidadesFiltradas.length; const entregues = unidadesFiltradas.filter(u => u.chaves === "Entregue").length; const pendentes = unidadesFiltradas.filter(u => u.situacao !== "Liberada" && u.situacao !== "Aprovada" ).length; const liberadas = unidadesFiltradas.filter(u => u.situacao === "Liberada").length; const aprovadas = unidadesFiltradas.filter(u => u.situacao === "Aprovada").length; const cvco = unidadesFiltradas.filter(u => u.cvco === "Liberado").length; // --- 2. ATUALIZAR CARDS --- document.querySelector("#cardTotal p").textContent = total; document.querySelector("#cardEntregues p").textContent = entregues; document.querySelector("#cardPendentes p").textContent = pendentes; document.querySelector("#cardLiberadas p").textContent = liberadas; document.querySelector("#cardAprovadas p").textContent = aprovadas; document.querySelector("#cardCVCO p").textContent = cvco; // --- 3. ATUALIZAR TABELA --- const tbody = document.getElementById("tabelaDash"); tbody.innerHTML = ""; if (total === 0) { tbody.innerHTML = 'Nenhuma unidade encontrada com estes filtros.'; return; } unidadesFiltradas.forEach(u => { const tr = document.createElement("tr"); tr.innerHTML = ` ${u.empreendimento} ${u.bloco} ${u.unidade} ${u.situacao} ${u.statusFinanceiro} ${u.habitavel} ${u.cvco} ${u.chaves} ${u.dataVistoria || '-'} ${u.dataLiberacao || '-'} `; tbody.appendChild(tr); }); }
